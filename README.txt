លេងរែក V10 — បន្ថែមច្បាប់ព័ទ្ធជាក្រុម

រក្សាទុក:
- ការរៀបកូន 16/16 តាមរូប។
- ការចេញដំបូងជ្រើសកូន 2 និងផ្លាស់ទីជាមួយគ្នា។
- ច្បាប់រែក V9 ដែលអ្នកបានបញ្ជាក់ថា 100% ត្រឹមត្រូវ។

ច្បាប់ព័ទ្ធ:
1. កូនដែលជាប់គ្នាតាម 4 ទិស (លើ/ក្រោម/ឆ្វេង/ស្តាំ) គឺជាក្រុមតែមួយ។
2. កូនខ្លួនឯងដែលជាប់គ្នា មិនមែនជាប្រឡោះទេ; វាជាសមាជិកក្រុម។
3. ប្រឡោះ = ក្រឡាទំនេរ ដែលជាប់ក្រុមតាម 4 ទិស។
4. បើក្រុមមានប្រឡោះទំនេរ សូម្បីតែ 1 -> ក្រុមទាំងមូលរស់។
5. បើគ្មានប្រឡោះទំនេរសោះ -> ក្រុមទាំងមូលស្លាប់។
6. គែមក្តារ/ក្រៅក្តារ មិនរាប់ជាប្រឡោះទេ។
7. ក្រុមនៅជ្រុងអាចរស់ ប្រសិនបើនៅសល់ក្រឡាទំនេរខាងក្នុង។
8. បើក្រុមដែលត្រូវព័ទ្ធមានស្តេច -> ភាគីនោះចាញ់ភ្លាម។
9. ព័ទ្ធត្រូវបានពិនិត្យបន្ទាប់ពីការដើរ និងរែក នៅក្រុមសត្រូវដែលជាប់នឹងក្រឡាដែលទើបដើរចូល។
10. មុខងារព័ទ្ធមិនសម្លាប់ក្រុមដែលនៅមានប្រឡោះទេ។
