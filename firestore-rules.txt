rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  match /users/{userId} {
   allow read: if request.auth != null && request.auth.uid == userId;
   allow create: if request.auth != null && request.auth.uid == userId
     && request.resource.data.coins == 0
     && request.resource.data.points == 0
     && request.resource.data.wins == 0
     && request.resource.data.losses == 0
     && request.resource.data.draws == 0
     && request.resource.data.totalGames == 0;
   allow update: if request.auth != null && request.auth.uid == userId
     && request.resource.data.coins == resource.data.coins
     && request.resource.data.points == resource.data.points
     && request.resource.data.wins == resource.data.wins
     && request.resource.data.losses == resource.data.losses
     && request.resource.data.draws == resource.data.draws
     && request.resource.data.totalGames == resource.data.totalGames;
   allow delete: if false;
  }
  match /transactions/{id} {
   allow read: if request.auth != null && resource.data.userId == request.auth.uid;
   allow create: if request.auth != null && request.resource.data.userId == request.auth.uid
    && request.resource.data.type == "topup" && request.resource.data.status == "pending"
    && request.resource.data.createdAt == request.time
    && ((request.resource.data.packageId=="coins_10000" && request.resource.data.coins==10000 && request.resource.data.amount==0.99)
    || (request.resource.data.packageId=="coins_50000" && request.resource.data.coins==50000 && request.resource.data.amount==4.50)
    || (request.resource.data.packageId=="coins_100000" && request.resource.data.coins==100000 && request.resource.data.amount==8.99)
    || (request.resource.data.packageId=="coins_500000" && request.resource.data.coins==500000 && request.resource.data.amount==30.99));
   allow update, delete: if false;
  }
  match /{document=**} { allow read, write: if false; }
 }
}
